public sealed class ConnectionRequest
{
    // [CORE]
    public long Id { get; private set; }

    // [CORE]
    // Direct ownership scope for My Requests and own request details.
    public long ClientAccountId { get; private set; }

    // [CORE]
    // Stored reference to ApplicantParty aggregate.
    public long ApplicantPartyId { get; private set; }

    // [CORE]
    public RequestStatus Status { get; private set; }

    // [CORE]
    public RequestDetails Details { get; private set; }

    // [CORE]
    // Request object location = object address.
    public ObjectAddress ObjectAddress { get; private set; }

    

    // [CORE]
    private ReviewDecisionRecord? _reviewDecision;

    // [EXT][VAR:EXPAND]
    // Request documents may be added without changing status transitions.
    private readonly List<RequestDocumentAttachment> _documents = [];

    // [DEFER]
    // Verification is request-context only.
    // Do not model standalone ApplicantData verification.
    private VerificationResult? _verificationResult;

    private ConnectionRequest() { }

    public static Result<ConnectionRequest> Create(
        long currentClientAccountId,
        ApplicantParty applicantParty,
        RequestApplicantData requestApplicantData,
        RequestDetails details,
        ObjectAddress objectAddress)
    {
        // [CORE]
        if (applicantParty is null)
            throw new ArgumentNullException(nameof(applicantParty));

        // [CORE]
        if (applicantParty.Id <= 0)
            throw new InvalidOperationException("ApplicantParty must be persisted.");

        // [CORE]
        // ApplicantParty must belong to current client.
        if (applicantParty.ClientAccountId != currentClientAccountId)
            return Result.Failure(RequestErrors.ApplicantPartyDoesNotBelongToClient);

        // [CORE]
        // Invalid request-local applicant data prevents request creation.
        if (requestApplicantData is null)
            return Result.Failure(RequestErrors.ApplicantDataRequired);

        // [CORE]
        if (details is null)
            return Result.Failure(RequestErrors.DetailsRequired);

        // [CORE]
        if (objectAddress is null)
            return Result.Failure(RequestErrors.ObjectAddressRequired);

        return new ConnectionRequest
        {
            ClientAccountId = currentClientAccountId,
            ApplicantPartyId = applicantParty.Id,
            ApplicantData = requestApplicantData,
            Details = details,
            ObjectAddress = objectAddress,

            // [CORE]
            // Target initial lifecycle state.
            Status = RequestStatus.InReview
        };
    }

    public bool CanStartReview()
    {
        // [CORE]
        return Status == RequestStatus.InReview;
    }

    public Result Approve(EmployeeRef reviewer)
    {
        // [CORE]
        if (reviewer is null)
            return Result.Failure(RequestErrors.ReviewerRequired);

        // [CORE]
        // Only InReview request can be approved.
        if (Status != RequestStatus.InReview)
            return Result.Failure(RequestErrors.OnlyInReviewCanBeApproved);

        Status = RequestStatus.Approved;

        // [CORE]
        // Decision is recorded atomically with status transition.
        _reviewDecision = ReviewDecisionRecord.Approved(reviewer);

        // [CORE]
        // Approval does not create agreement proposal.
        // Proposal exchange starts later by employee action.

        return Result.Success();
    }

    public Result Reject(EmployeeRef reviewer, RejectionFeedback feedback)
    {
        // [CORE]
        if (reviewer is null)
            return Result.Failure(RequestErrors.ReviewerRequired);

        // [CORE]
        // Only InReview request can be rejected.
        if (Status != RequestStatus.InReview)
            return Result.Failure(RequestErrors.OnlyInReviewCanBeRejected);

        // [CORE]
        // Rejection feedback is visible to client.
        if (feedback is null || feedback.IsEmpty)
            return Result.Failure(RequestErrors.RejectionFeedbackRequired);

        Status = RequestStatus.Rejected;

        // [CORE]
        _reviewDecision = ReviewDecisionRecord.Rejected(reviewer, feedback);

        return Result.Success();
    }

    public Result AttachDocument(RequestDocumentAttachment attachment)
    {
        // [EXT]
        // Future request document flow.
        // Domain stores accepted metadata/reference, not file/blob.
        if (attachment is null)
            return Result.Failure(RequestErrors.DocumentRequired);

        _documents.Add(attachment);
        return Result.Success();
    }

    public Result RecordVerificationResult(VerificationResult result)
    {
        // [DEFER]
        // Verification is available only in request context.
        if (result is null)
            return Result.Failure(RequestErrors.VerificationResultRequired);

        _verificationResult = result;
        return Result.Success();
    }
