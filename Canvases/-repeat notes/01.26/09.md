using System.Linq;
using System.Text;
using System.Text.RegularExpressions;
public class StripCommentsSolution
{
    public static string StripComments(string text, string[] commentSymbols)
    {
        var sb= new StringBuilder();
        var noNL = text.Split('\n');
        var charSep = commentSymbols.Select(s=>s[0]).ToArray();
        var strArr= noNL.Select(str=>str.Split(charSep)[0].TrimEnd())
          .Where(str=>str!=null).ToList();
        for(int i=0;i<strArr.Count;i++){
          sb.Append(strArr[i]);
          if(i!=strArr.Count-1){
            sb.Append('\n');
          }
        }
        return sb.ToString();
    }
} ^aQnY7ZjF
