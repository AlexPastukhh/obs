public class Solution {
    public int MaxCount =0;
    public Dictionary<char,int> CharDict = new Dictionary<char,int>();
    public Dictionary<int,char> IndexDict = new Dictionary<int,char>();
    public int LengthOfLongestSubstring(string s) {
        for(int i =0;i<s.Length;i++){
            if(CharDict.ContainsKey(s[i])){
                TryNewSubstring(s[i],i);
            }
            else{
                AddNewChar(s[i],i);
            }

            if(i==s.Length-1){
                ChangeMaxCountIfNeeded();
            }

        }
        return MaxCount;
    }
    public void ChangeMaxCountIfNeeded(){
        if(CharDict.Count()>MaxCount){
            MaxCount = CharDict.Count();
        }
        
    }
    public void AddNewChar(char newChar,int index){
        CharDict.Add(newChar,index);
        IndexDict.Add(index,newChar);
    }
    public void TryNewSubstring(char repeated,int ind){
        ChangeMaxCountIfNeeded();
        var index = CharDict[repeated];
        for(;index>=0;index--){
            if(IndexDict.TryGetValue(index,out char value)){
                IndexDict.Remove(index);
                CharDict.Remove(value);
            }
            
        }
        AddNewChar(repeated,ind);
    }

} ^nFBdFXSZ
