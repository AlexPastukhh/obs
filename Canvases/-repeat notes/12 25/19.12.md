import { useQuery } from "@tanstack/react-query"
import { fetchRepos, PAGE_SIZE } from "./api"

function RepoList() {
  const [page, setPage] = useState(1)
  const [sort, setSort] = useState("stars")
  
  const { data, isLoading, isFetching } = useQuery({
    queryKey: ['repos', { sort, page }],
    queryFn: () => fetchRepos(sort, page),
    // Keep previous data for smooth pagination
    placeholderData: (previousData) => previousData
  })
  
  // Calculate if we're at the beginning or end
  const isFirstPage = page === 1
  const isLastPage = data?.length < PAGE_SIZE
  
  // Could also be: data?.length === 0 (empty page)
  
  return (
    <div>
      <SortControls sort={sort} onChange={setSort} />
      
      {data?.map(repo => (
        <RepoItem key={repo.id} repo={repo} />
      ))}
      
      <div className="pagination">
        <button 
          onClick={() => setPage(p => Math.max(1, p - 1))}
          disabled={isFirstPage || isFetching}
        >
          Previous
        </button>
        
        <span>Page {page}</span>
        
        <button 
          onClick={() => setPage(p => p + 1)}
          disabled={isLastPage || isFetching}
        >
          Next
        </button>
        
        {isFetching && <span>Loading...</span>}
      </div>
    </div>
  )
} ^y8cFDsrY
