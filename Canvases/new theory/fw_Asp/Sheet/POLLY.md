static TimeSpan ComputeDelay(HttpResponseMessage response, int attempt)
{
    // 1) Use Retry-After if present
    if (response.Headers.RetryAfter is { } ra)
    {
        if (ra.Delta is TimeSpan delta)
            return Clamp(delta);

        if (ra.Date is DateTimeOffset date)
            return Clamp(date - DateTimeOffset.UtcNow);
    }

    // 2) Otherwise use exponential backoff + jitter
    var baseDelay = TimeSpan.FromMilliseconds(200 * Math.Pow(2, attempt - 1)); // 200ms, 400ms, 800ms...
    var jitter = TimeSpan.FromMilliseconds(Random.Shared.Next(0, 200));
    return Clamp(baseDelay + jitter);
}

static TimeSpan Clamp(TimeSpan delay)
{
    if (delay < TimeSpan.Zero) return TimeSpan.Zero;
    // Protect yourself from crazy Retry-After values
    var max = TimeSpan.FromSeconds(30);
    return delay > max ? max : delay;
}
