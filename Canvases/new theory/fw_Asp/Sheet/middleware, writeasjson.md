    private readonly ILogger<RequestTimingMiddleware> _log;

    public RequestTimingMiddleware(RequestDelegate next, ILogger<RequestTimingMiddleware> log)
    {

        _log = log;
    }

    public async Task InvokeAsync(HttpContext context)
    {
        var sw = Stopwatch.StartNew();
        await _next(context);
        sw.Stop();

        context.Response.Headers["X-Elapsed-ms"] = sw.ElapsedMilliseconds.ToString();
        _log.LogInformation("Elapsed: {ms}ms", sw.ElapsedMilliseconds);
    }
}
