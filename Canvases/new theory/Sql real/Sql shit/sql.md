-- DEMONSTRATING MASTER, MODEL, and GO

-- 1. Show we're in MASTER (control database)
USE master;

SELECT name, state_desc, create_date 
FROM sys.databases

GO

-- 3. Look at MODEL's current settings
SELECT 
    'MODEL Database Settings:',
    name,
    size/128.0 AS SizeMB,
    CASE max_size 
        WHEN -1 THEN 'UNLIMITED' 
        ELSE CAST(max_size/128.0 AS VARCHAR) + ' MB' 
    END AS MaxSize
FROM sys.master_files 
WHERE database_id = DB_ID('model');
GO

GO  -- Batch separator required!

-- 5. Switch to new database
USE DemoDB;
GO  -- Required after USE!

-- 6. Check what was created (inherited from MODEL)

GO

-- 7. Clean up
USE master;
GO
