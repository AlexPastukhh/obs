WITH AggData AS (
    -- Step 1: Pre-aggregate
    SELECT
        Product,
        Region,
        Quarter,
        SUM(SalesAmount)  AS TotalSales,
        COUNT(*)          AS SalesCount,
        AVG(SalesAmount)  AS AvgSales
    FROM Sales
    GROUP BY Product, Region, Quarter
),
Unpivoted AS (
    -- Step 2: Convert metrics into rows
    SELECT
        Product,
        Region,
        Quarter,
        Metric,
        Value
    FROM AggData
    UNPIVOT
    (
        Value FOR Metric IN (TotalSales, SalesCount, AvgSales)
    ) u
),
PrepForPivot AS (
    -- Step 3: Build pivot column names
    SELECT
        Product,
        Region,
        CONCAT(Quarter, '_', Metric) AS PivotCol,
        Value
    FROM Unpivoted
)
-- Step 4: Pivot to final report
SELECT
    Product,
    Region,
    [Q1_TotalSales],
    [Q1_SalesCount],
    [Q1_AvgSales],
    [Q2_TotalSales],
    [Q2_SalesCount],
    [Q2_AvgSales]
FROM PrepForPivot
PIVOT
(
    MAX(Value)
    FOR PivotCol IN (
        [Q1_TotalSales],
        [Q1_SalesCount],
        [Q1_AvgSales],
        [Q2_TotalSales],
        [Q2_SalesCount],
        [Q2_AvgSales]
    )
) p
ORDER BY Product, Region;
