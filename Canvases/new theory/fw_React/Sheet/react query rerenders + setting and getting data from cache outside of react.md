// src/api/fetchWithAuth.ts
import { getAccessToken, setAccessToken } from "../auth/tokenCache";
import { refreshAccessToken } from "../auth/refresh";

let refreshPromise: Promise<string> | null = null;

async function refreshOnce(): Promise<string> {
  if (!refreshPromise) {
    refreshPromise = refreshAccessToken().finally(() => {
      refreshPromise = null;
    });
  }
  return refreshPromise;
}

export async function fetchWithAuth(input: RequestInfo, init: RequestInit = {}) {
  const headers = new Headers(init.headers);

  const token = getAccessToken(); // <-- read from React Query cache
  if (token) headers.set("Authorization", `Bearer ${token}`);

  const doReq = () =>
    fetch(input, { ...init, headers, credentials: "include" });

  let res = await doReq();
  if (res.status !== 401) return res;

  // 401 -> refresh -> retry once
  try {
    const fresh = await refreshOnce();
    headers.set("Authorization", `Bearer ${fresh}`);
    res = await doReq();
    return res;
  } catch {
    // Refresh failed: clear token so UI switches to login
    setAccessToken(null);
    return res; // still 401
  }
} ^XPl4d6dM

import { setAccessToken } from "./tokenCache";

export async function login(username: string, password: string) {
  const res = await fetch("/api/auth/login", {
    method: "POST",
    credentials: "include",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ username, password }),
  });

  if (!res.ok) throw new Error(await res.text());

  const data = (await res.json()) as { accessToken: string };
  setAccessToken(data.accessToken); // updates cache -> UI re-renders
} ^5KJHerPG
