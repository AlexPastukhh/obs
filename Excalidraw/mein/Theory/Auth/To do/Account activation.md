        {
            Guard.IsNotNull(inputCode);
            Guard.IsNotNull(utcNow);

            if (inputCode != SecurityCode)
            {
                IsActivationAttempted = true;
                return Errors.Account.InValidActivationCode(SecurityCode, inputCode);

            if(utcNow > CodeExpirationTime)
            {

            }

            IsActivationAttempted = true;
            IsActivated = true;
