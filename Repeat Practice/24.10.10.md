   ```
   defineSlots<{
      itemNameSlot: (props: { itemName: string }) => any;
    }>();
    </script>

    // ----------------------------

    <!-- <ParentComponent> -->

      <main>
       <Test>
         <template #itemNameSlot="{ itemName }">
           <a :href="`/someurl/${itemName}`"> 
             {{ itemName }} - click 
           </a>
         </template>
       </Test>
      </main>
    </template>
```
