1 ensure errors cause everything can be thrown 
`function ensureError(value: unknown): Error {  
`	if (value instanceof Error) return value  `
`	let stringified = '[Unable to stringify the thrown value]'  `
`	try {  `
`	stringified = JSON.stringify(value)  `
`	} catch {}  `
``	const error = new Error(`This value was thrown as is, not through an     Error: ${stringified}`)  ``
`	return error  `
`}``
2 send cause of error to not lose stack trace 
const error1 = new Error("Network error")  
const error2 = new Error("The update failed", { cause: error1 })
3 use constant error messages for error profilers that will perceive same error with different messages as different errors 
4 explicitly show that something can return error or failed result
5 dont add too much context
