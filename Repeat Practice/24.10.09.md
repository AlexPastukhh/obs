1 provide inject
#####  To provide some data or methods down the component tree you can do:
// in parent 
	provide("somename",somedata)
// in child 
	const somename = inject("somename")
	

##### Can also provide reactive data, but to DRY, you need to <mark style="background: #FF5582A6;">keep all changes of your data inside your component.</mark> It means you need to provide method which will change this reactive data

// in parent 
	const methodToChangeData=()=>{
	}
	provide("somename",{
	somedata,
	methodToChangeData
	})
	
// in child 
	const { somedata, methodToChangeData } = inject('somename')
#### If value wasnt provided by parent, then you can use <mark style="background: #FF5582A6;">default value</mark>
// in child 
	const somename = inject("somename",defaultValue)
// If default value takes some price to create then can use third attribute true with factory func
	const somename = inject("somename",()=>new ExpensiveClass(),true)
#####    In larger applications you can use keys file with symbols 

// keys.js 
const somename = Symbol("description for debugging");
// in parent 
	import somename from 'keys.js'
	provide(somename,somedata)
// in child 
	import somename from 'keys.js'
	const somename = inject(somename)

24 v-model / defineModel

`<template>
`  <button @click="OnClick">Increment {{props.modelValue}}</button>`
`</template>`

`<script setup>`
`import {defineProps, defineEmits } from 'vue';`
`const props = defineProps(['modelValue']);`
`const emit = defineEmits(['update:modelValue']);`
`const OnClick = ()=>{`
	`  const newValue = props.modelValue + 1;`
	`  emit("update:modelValue",newValue);`
`}`
`</script>``
