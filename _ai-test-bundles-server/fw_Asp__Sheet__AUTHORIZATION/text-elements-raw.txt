using Microsoft.AspNetCore.Authorization;
using System.Security.Claims;

public sealed class ScopeRequirementHandler : AuthorizationHandler<ScopeRequirement>
{
    protected override Task HandleRequirementAsync(AuthorizationHandlerContext context, ScopeRequirement requirement)
    {
        if (HasScope(context.User, requirement.Scope))
            context.Succeed(requirement);

        return Task.CompletedTask; // do nothing => not satisfied
    }

    private static bool HasScope(ClaimsPrincipal user, string needed)
    {
        // gather all possible scope strings
        var values = user.FindAll("scope").Select(c => c.Value)
            .Concat(user.FindAll("scp").Select(c => c.Value));

        foreach (var v in values)
        {
            // handle both "a b c" and single-value
            var scopes = v.Split(' ', StringSplitOptions.RemoveEmptyEntries);
            if (scopes.Contains(needed, StringComparer.Ordinal))
                return true;
        }

        return false;
    }
} ^jcqM0nwN
