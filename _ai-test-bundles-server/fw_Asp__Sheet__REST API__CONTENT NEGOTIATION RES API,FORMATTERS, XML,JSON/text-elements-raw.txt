using Microsoft.AspNetCore.Mvc;
using Microsoft.Net.Http.Headers;

[HttpGet("{authorId}", Name = "GetAuthor")]
public async Task<IActionResult> GetAuthor(Guid authorId, string? fields,
    [FromHeader(Name = "Accept")] string? mediaType)
{
    // 1) Parse accept header
    if (!MediaTypeHeaderValue.TryParse(mediaType, out var parsedMediaType))
    {
        return BadRequest("Invalid Accept header.");
    }

    // 2) Fetch
    var authorFromRepo = await _courseLibraryRepository.GetAuthorAsync(authorId);
    if (authorFromRepo == null) return NotFound();

    var authorDto = _mapper.Map<AuthorDto>(authorFromRepo);

    // 3) Representation decision
    if (parsedMediaType.MediaType == "application/vnd.marvin.hateoas+json")
    {
        var links = CreateLinksForAuthor(authorId, fields);

        var shaped = authorDto.ShapeData(fields) as IDictionary<string, object?>;
        shaped!.Add("links", links);

        return Ok(shaped);
    }

    // default/plain representation
    return Ok(authorDto.ShapeData(fields));
} ^8yG1NObz
