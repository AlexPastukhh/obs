    {
        var emailStr = _contextAccessor.HttpContext!.User.Claims.
                FirstOrDefault(c => c.Type == ClaimTypes.Email)!.Value;

        
        var protector = _protectionProvider.CreateProtector(

            return sendLink.Error;

            "Reset password",

            cancellation);

            return sendEmail.Error;
        }
