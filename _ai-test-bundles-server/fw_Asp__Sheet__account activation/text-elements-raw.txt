using Microsoft.AspNetCore.DataProtection;
using Microsoft.AspNetCore.WebUtilities;
using System.Text;

public static class ActivationTokens
{
    public static string Create(IDataProtector protector, string userId, DateTimeOffset expiresUtc)
    {
        var payload = $"{userId}|{expiresUtc.UtcDateTime:o}";
        var protectedBytes = protector.Protect(Encoding.UTF8.GetBytes(payload));
        return WebEncoders.Base64UrlEncode(protectedBytes);
    }

    public static bool TryValidate(IDataProtector protector, string token, out string userId)
    {
        userId = "";
        byte[] protectedBytes;
        try { protectedBytes = WebEncoders.Base64UrlDecode(token); }
        catch { return false; }

        byte[] bytes;
        try { bytes = protector.Unprotect(protectedBytes); }
        catch { return false; } // tampered / wrong key / etc.

        var payload = Encoding.UTF8.GetString(bytes);
        var parts = payload.Split('|');
        if (parts.Length != 2) return false;

        userId = parts[0];
        if (!DateTimeOffset.TryParse(parts[1], out var exp)) return false;

        return exp > DateTimeOffset.UtcNow;
    }
} ^2Yh5uI3s

    HttpContext http,
    RegisterRequest req) =>
{

    var result = await userManager.CreateAsync(user, req.Password);
    if (!result.Succeeded) return Results.BadRequest(result.Errors);

    var token = await userManager.GenerateEmailConfirmationTokenAsync(user);
