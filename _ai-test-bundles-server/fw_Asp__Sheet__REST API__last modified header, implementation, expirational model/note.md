using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

[ApiController]
[Route("api/authors")]
public class AuthorsController : ControllerBase
{
    private readonly AppDbContext _db;
    public AuthorsController(AppDbContext db) => _db = db;

    [HttpGet("{id:guid}")]
    public async Task<IActionResult> Get(Guid id, CancellationToken ct)
    {
        var data = await _db.Authors.AsNoTracking()
            .Where(a => a.Id == id)
            .Select(a => new
            {
                a.Id,
                a.Name,
                LastModified = EF.Property<DateTimeOffset>(a, "LastModified")
            })
            .SingleOrDefaultAsync(ct);

        if (data is null) return NotFound();

        Response.Headers.LastModified = data.LastModified.UtcDateTime.ToString("R");

        // Basic If-Modified-Since check (seconds precision)
        if (Request.Headers.IfModifiedSince.Count > 0 &&
            DateTimeOffset.TryParse(Request.Headers.IfModifiedSince.ToString(), out var ims))
        {
            var lmSeconds = new DateTimeOffset(
                data.LastModified.UtcDateTime.AddTicks(-(data.LastModified.UtcDateTime.Ticks % TimeSpan.TicksPerSecond)),
                TimeSpan.Zero);

            if (lmSeconds <= ims.ToUniversalTime())
                return StatusCode(StatusCodes.Status304NotModified);
        }

        return Ok(new { data.Id, data.Name });
    }
